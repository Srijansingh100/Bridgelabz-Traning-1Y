import java.util.*;
class cwarray2 {
	public static void main (String args[]){
	int arr[]=new int[5];
		arr[0]=56;
		arr[1]=67;
		arr[2]=90;
		arr[3]=43;
		arr[4]=108;
		
		System.out.print("printing array with to string : ");
		System.out.println(Arrays.toString(arr));
	
		System.out.print("length of an array : ");
		System.out.println(arr.length);
		
		System.out.print("Print an array using for loop : ");
		for(int i=0;i<arr.length;i++){
			System.out.println(arr[i]);
		}
		
		System.out.print("Print an array with for each : ");
		for(int i : arr){
			System.out.println(i);
		}
		
		int sum = 0;
		System.out.print("sum of array is : ");
		
		for(int i=0;i<arr.length;i++){
			sum = sum + arr[i];
			
		}
		System.out.println(sum);
		System.out.print("reversed array is : ");
		
		for(int i = 4;i>=0;i--){
			System.out.println(arr[i]);
		}
		int min = arr[0];
		int max = arr[0];
		for(int i = 0;i<=4;i++){
			if(arr [i] < min){
				min = arr[i];
			}
			if(arr [i] > max){
				max = arr[i];
			}
		}
		sum = max + min;
		System.out.println("minimum no. : "+min);
		System.out.println("maximum no. : "+max);
		System.out.println("sum of minimum array and maximum array : "+sum);
	}
}
